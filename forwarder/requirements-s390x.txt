# Workload Security — Forwarder Constraints for linux/s390x
# VERIFIED on IBM LinuxONE (s390x) on 2026-10-18.

# Install with:
#   pip install --only-binary=:all: -r forwarder/requirements-s390x.txt
#
# --only-binary is mandatory. Every package below resolves to a prebuilt
# s390x wheel; without the flag pip may attempt a source build, which is
# slow on this target and unnecessary.

# ── Core runtime (verified: all pure Python, no compilation) ───────────

requests==2.31.0
pytest==8.1.1

# ── Dashboard API runtime ──────────────────────────────────────────────
# httpx MUST stay < 0.28. fastapi 0.110.0 pins starlette 0.36.3, whose
# TestClient passes `app=` to httpx.Client; httpx 0.28 removed that kwarg
# and every TestClient-based test errors with
#   TypeError: Client.__init__() got an unexpected keyword argument 'app'
# This is not optional. An unpinned httpx silently breaks the test suite.
httpx==0.27.2

# starlette's WebSocket implementation needs a native ws library.
# Wheels confirmed for s390x.
websockets==17.2

# main.py imports these at module level, so the package will not import
# without them even on the offline path.
slowapi==0.1.10
python-dotenv==1.2.4

# Pure Python; confirmed manylinux_2_17_s390x wheels.
PyYAML==6.0.3

# Needed by remediation/metrics.py; without it the remediation agent fails
# to import and /api/health reports it as unavailable.
prometheus-client==0.26.0

# ── Test tooling ───────────────────────────────────────────────────────
# pytest-asyncio must stay on a 0.23.x line: 1.x forces pytest>=8.2 and
# pulls the suite off the pinned pytest.
pytest-asyncio==0.23.5
